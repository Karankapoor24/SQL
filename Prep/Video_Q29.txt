/*
PROBLEM STATEMENT: Given table provides login and logoff details of one user.
Generate a report to reqpresent the different periods (in mins) when user was logged in.
*/

create table login_details
(
	times	time,
	status	varchar(3)
);

Output


log_on				log_off				duration
10:00:00.0000000	10:03:00.0000000	3
10:04:00.0000000	10:06:00.0000000	2
10:09:00.0000000	10:13:00.0000000	4
10:15:00.0000000	10:16:00.0000000	1


Solution


WITH cte AS (
SELECT
	*, ROW_NUMBER() OVER(ORDER BY times) AS rn
FROM login_details
), cte_grp AS (
SELECT 
	times, status, rn, rn - ROW_NUMBER() OVER(ORDER BY times) AS grp
FROM cte
WHERE status = 'on'
)
SELECT
	MIN(times) AS log_on, DATEADD(MINUTE, 1,  MAX(times)) AS log_off, COUNT(times) AS duration
FROM cte_grp
GROUP BY grp
