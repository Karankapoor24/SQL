-- 2 Friends2 have strong Friends2hip if they have more than or equal to 3 mutual Friends2 between them.

CREATE TABLE Friends2 (
    User1 INT NOT NULL,
    User2 INT NOT NULL
);


INSERT INTO Friends2 (User1, User2) VALUES (1, 2);
INSERT INTO Friends2 (User1, User2) VALUES (1, 4);
INSERT INTO Friends2 (User1, User2) VALUES (1, 5);
INSERT INTO Friends2 (User1, User2) VALUES (2, 4);
INSERT INTO Friends2 (User1, User2) VALUES (2, 5);
INSERT INTO Friends2 (User1, User2) VALUES (3, 1);
INSERT INTO Friends2 (User1, User2) VALUES (3, 2);
INSERT INTO Friends2 (User1, User2) VALUES (4, 5);


-- Input

User1   User2
1		2
1		4
1		5
2		4
2		5
3		1
3		2
4		5


-- Output

user1	user2	friends_list	friends_cnt
1		2		4,5,3			3
1		4		5,2				2
1		5		2,4				2
2		4		5,1				2
2		5		1,4				2
3		1		2				1
3		2		1				1
4		5		1,2				2



-- Solution


with friendships as (
select user1, user2 from friends2
union all
select user2, user1 from friends2
)
select a.user1, a.user2, string_agg(b.user2, ',') friends_list, count(*) as friends_cnt from friends2 a
left join friendships b
ON a.user1 = b.user1 
left join friendships c
ON a.user2 = c.user1 
where b.user2 = c.user2
group by a.user1, a.user2
order by 1,2
